select s.*,a.nom_art,a.ref_art,a.code_art,ca.id_cat,ca.nom_cat,a.seuil_art,m.nom_mag,
                p.prix_mini_art,p.prix_gros_art,p.prix_max_art
                from t_stock s
   inner join t_magasin m on s.mag_stk=m.id_mag
  inner join t_article a on s.art_stk=a.id_art
  inner join t_categorie_article ca on a.cat_art=ca.id_cat
  left join (select * from t_prix_article GROUP BY art_prix_art DESC) p on s.art_stk=p.art_prix_art
  WHERE 1=1 AND s.mag_stk=10 Order by m.nom_mag,ca.nom_cat,a.nom_artSELECT *,time(Date_vnt) as heure_vnt FROM v_etat_ventes WHERE sup_fact=0 AND bl_fact_crdt=0  AND id_mag=2 AND date(Date_vnt) between '2023-07-21' 
                AND '2023-07-21' ORDER BY code_mag,Date_vnt DESC,nom_cat ASC
